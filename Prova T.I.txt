CREATE DATABASE HOTEL;

USE HOTEL;

CREATE TABLE QUARTOS (
    ID INT PRIMARY KEY,
    NUMERO VARCHAR(50),
    CAPACIDADE INT,
    VISTA_MAR BOOLEAN,
    DIARIA DECIMAL(10,2)
);

CREATE TABLE CLIENTES (
    ID INT PRIMARY KEY,
    NOME VARCHAR(100),
    EMAIL VARCHAR(100),
    TELEFONE VARCHAR(20),
    TABELA VARCHAR(100)
);

CREATE TABLE RESERVAS (
    ID INT PRIMARY KEY,
    CLIENTE_ID INT,
    QUARTO_ID INT,
    DATA_ENTRADA DATE,
    DATA_SAIDA DATE,
    FOREIGN KEY (CLIENTE_ID) REFERENCES CLIENTES(ID),
    FOREIGN KEY (QUARTO_ID) REFERENCES QUARTOS(ID)
);

3)
 A) Quartos disponíveis com vista para o mar
SELECT *
FROM QUARTOS
WHERE VISTA_MAR = TRUE;

 B) Nome e e-mail de todos os clientes
SELECT NOME, EMAIL
FROM CLIENTES;

 C) Quartos com capacidade para mais de 2 pessoas
SELECT *
FROM QUARTOS
WHERE CAPACIDADE > 2;

 D) Clientes que começam com a letra B
SELECT *
FROM CLIENTES
WHERE NOME LIKE 'B%';

 E) Reservas realizadas entre 01/05/2023 e 31/05/2023
SELECT *
FROM RESERVAS
WHERE DATA_ENTRADA BETWEEN '2023-05-01' AND '2023-05-31';

4)
 A) Total de quartos
SELECT COUNT(*) AS TOTAL_QUARTOS
FROM QUARTOS;

 B) Quarto com menor capacidade
SELECT *
FROM QUARTOS
WHERE CAPACIDADE = (SELECT MIN(CAPACIDADE) FROM QUARTOS);

 C) Quarto com maior capacidade
SELECT *
FROM QUARTOS
WHERE CAPACIDADE = (SELECT MAX(CAPACIDADE) FROM QUARTOS);

 D) Média de capacidade dos quartos
SELECT AVG(CAPACIDADE) AS MEDIA_CAPACIDADE
FROM QUARTOS;

 E) Total de quartos com vista para o mar
SELECT COUNT(*) AS TOTAL_QUARTOS_VISTA_MAR
FROM QUARTOS
WHERE VISTA_MAR = TRUE;

5)
 A) Quartos por diária em ordem decrescente
SELECT *
FROM QUARTOS
ORDER BY DIARIA DESC;

 B) Quartos por diária em ordem crescente
SELECT *
FROM QUARTOS
ORDER BY DIARIA ASC;
